### UC97_TC1.functions.js
```javascript
import { expect } from '@playwright/test';
import { TodoListPage } from '../../models/page_object_models/todo_list_page.js';

export const openTodoPage = async function(page, reporter) {
    const startTime = new Date().getTime();
    const todoPage = new TodoListPage(page);

    await todoPage.open();
    await expect(todoPage.newTaskInput).toBeVisible({ timeout: 10000 });

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC97_TC1_ID1', 'Apre la pagina delle attività', 'Il campo per aggiungere un\'attività è visibile', 'Campo attività visibile', true, '', executionTime);
    }
};

export const addTask = async function(page, reporter) {
    const startTime = new Date().getTime();
    const todoPage = new TodoListPage(page);

    await todoPage.addTask('Buy milk');
    await expect(todoPage.taskItems.filter({ hasText: 'Buy milk' }).first()).toBeVisible({ timeout: 10000 });

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC97_TC1_ID2', 'Inserisce e aggiunge l\'attività', 'L\'attività compare nella lista', 'Attività Buy milk visibile', true, '', executionTime);
    }
};

export const checkTaskList = async function(page, reporter) {
    const startTime = new Date().getTime();
    const todoPage = new TodoListPage(page);

    await expect(todoPage.taskItems.first()).toBeVisible({ timeout: 10000 });
    const taskCount = await todoPage.taskItems.count();

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC97_TC1_ID3', 'Controlla la lista', 'La lista contiene almeno un\'attività', `Attività visibili: ${taskCount}`, true, '', executionTime);
    }
};
```
