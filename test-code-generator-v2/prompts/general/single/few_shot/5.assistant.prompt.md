### UC99_TC2.functions.js
```javascript
import { expect } from '@playwright/test';
import { WarehousePage } from '../../models/page_object_models/warehouse_page.js';

export const openWarehousePage = async function(page, reporter) {
    const startTime = new Date().getTime();
    const warehousePage = new WarehousePage(page);

    await warehousePage.open();
    await expect(warehousePage.statusFilter).toBeVisible({ timeout: 10000 });

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC99_TC2_ID1', 'Apre la pagina del magazzino', 'La pagina mostra il filtro di stato', 'Filtro di stato visibile', true, '', executionTime);
    }
};

export const applyEmptyFilter = async function(page, reporter) {
    const startTime = new Date().getTime();
    const warehousePage = new WarehousePage(page);

    await warehousePage.applyFilter();
    await expect(warehousePage.errorMessage).toBeVisible({ timeout: 10000 });

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC99_TC2_ID2', 'Applica il filtro senza stato', 'Viene mostrato un messaggio di errore', 'Messaggio di errore visibile', true, '', executionTime);
    }
};

export const checkErrorMessage = async function(page, reporter) {
    const startTime = new Date().getTime();
    const warehousePage = new WarehousePage(page);

    await expect(warehousePage.errorMessage).toContainText('status', { timeout: 10000 });
    const message = await warehousePage.errorMessage.innerText();

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC99_TC2_ID3', 'Legge il messaggio di errore', 'Il messaggio chiede di selezionare uno stato', message, true, '', executionTime);
    }
};
```
