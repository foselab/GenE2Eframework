### UC99_TC1.functions.js
```javascript
import { expect } from '@playwright/test';
import { WarehousePage } from '../../models/page_object_models/warehouse_page.js';
import TestResultReporter from '../../models/test-result-reporter.js';

export const openWarehousePage = async function(page, reporter) {
    const startTime = new Date().getTime();
    const warehousePage = new WarehousePage(page);

    await warehousePage.open();
    await expect(warehousePage.statusFilter).toBeVisible({ timeout: 10000 });

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC99_TC1_ID1', 'Apre la pagina del magazzino', 'La pagina mostra il filtro di stato', 'Filtro di stato visibile', true, '', executionTime);
    }
};

export const filterByStatus = async function(page, reporter) {
    const startTime = new Date().getTime();
    const warehousePage = new WarehousePage(page);

    await warehousePage.selectStatus('open');
    await warehousePage.applyFilter();
    await expect(warehousePage.statusFilter).toHaveValue('open');

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC99_TC1_ID2', 'Seleziona lo stato e applica il filtro', 'Il filtro è impostato su open', 'Filtro impostato su open', true, '', executionTime);
    }
};

export const checkRowsAreVisible = async function(page, reporter) {
    const startTime = new Date().getTime();
    const warehousePage = new WarehousePage(page);

    await expect(warehousePage.rows.first()).toBeVisible({ timeout: 10000 });
    const rowCount = await warehousePage.rows.count();

    const executionTime = (new Date().getTime() - startTime) / 1000;
    if (reporter) {
        reporter.addStep('UC99_TC1_ID3', 'Controlla le righe elencate', 'Almeno una riga è visibile', `Righe visibili: ${rowCount}`, true, '', executionTime);
    }
};
```

import TestResultReporter from '../../models/test-result-reporter.js';
